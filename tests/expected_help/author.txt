usage: mosdat author [-h] [--url URL]
                     {vms,doctor,start,session,capture,localize,verify,action,click,prompt-click,prompt-hover,prompt-type,hover,type,key,wait,shell,launch,validate,export,step,close}
                     ...

positional arguments:
  {vms,doctor,start,session,capture,localize,verify,action,click,prompt-click,prompt-hover,prompt-type,hover,type,key,wait,shell,launch,validate,export,step,close}
    vms                 List configured VMs and Proxmox power state
    doctor              Check author dashboard readiness
    start               Create an authoring session
    session             Show authoring session state
    capture             Capture the current VNC screen
    localize            Find a prompt on the current screen
    verify              Ask the VLM a yes/no screen question
    action              Run a confirmed authoring action
    click               Run a confirmed click action
    prompt-click        Localize a prompt and click its center
    prompt-hover        Localize a prompt and hover its center
    prompt-type         Localize a prompt, click it, then type text
    hover               Run a confirmed hover action
    type                Run a confirmed text input action
    key                 Run a confirmed key press action
    wait                Append and run a bounded wait action
    shell               Run a confirmed shell action
    launch              Run a confirmed launch action
    validate            Validate current draft scenario
    export              Export current draft scenario YAML
    step                Append or replace draft scenario steps
    close               Close an authoring session

options:
  -h, --help            show this help message and exit
  --url URL             Live dashboard URL
