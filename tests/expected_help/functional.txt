usage: mosdat functional [-h] --vms VMS [--test NAME] [--model MODEL]
                         [--verify-model MODEL] [--save-screenshots]
                         [--screenshots DIR] [--popup-sweep] [--from-step N]
                         [--until-step N] [--skip-health-probe]
                         [--no-checkpoints] [--skip-warmup]
                         [--skip-workspace-check] [--skip-model-check]
                         [--timeout SECONDS] [--record] [--output OUTPUT]
                         [--click-verify {auto,off,yesno,diff,diff+yesno}]
                         [--canary {auto,off,on}] [--no-record-session]
                         [--record-fps FPS] [--record-diff-threshold VALUE]
                         [--record-gif] [--record-keep-raw]
                         [--inject-config JSON] [--inject-servers LIST]
                         [--inject-app-name NAME] [--inject-install-path PATH]
                         [--inject-migrations-version VER] [--no-cache]
                         [--config-snapshots] [--var KEY=VALUE]
                         [--from-phase ID] [--until-phase ID]
                         config

positional arguments:
  config                Path to mosdat config (TOML)

options:
  -h, --help            show this help message and exit
  --vms VMS             Comma-separated VM names
  --test NAME           Test file name without .yaml (default: rocketchat-
                        smoke)
  --model MODEL         Override VLM for element localization (default:
                        holo2-4b)
  --verify-model MODEL  Override VLM for yes/no state verification (default:
                        same as --model; recommend qwen3-vl-abliterated —
                        localization models hallucinate on yes/no)
  --save-screenshots    Save screenshots on failure to results dir
  --screenshots DIR     Save all step screenshots to this directory
  --popup-sweep         B3: sweep modal popups/dialogs before each localize
                        step
  --from-step N         B4: start from step N (1-indexed, default: 1)
  --until-step N        B4: stop after step N (inclusive, default: last step)
  --skip-health-probe   B7: skip VM health probe before scenario start
  --no-checkpoints      C2: disable Proxmox snapshot checkpoints even if YAML
                        enables them
  --skip-warmup         Skip VLM warmup phase (faster startup; first scenario
                        step may be slow if endpoint cold)
  --skip-workspace-check
                        Skip workspace URL preflight (faster startup; scenario
                        will burn VLM budget if server is down)
  --skip-model-check    H2.3: skip VLM model identity check (bypass
                        expected_model enforcement)
  --timeout SECONDS     I8: wall-clock watchdog — abort with exit 5 if
                        scenario exceeds N seconds (default: 900; 0 =
                        disabled; POSIX/Linux only — uses SIGALRM)
  --record              C3: interactive authoring mode — open VNC viewer,
                        capture clicks, generate YAML
  --output OUTPUT       (--record) path to write the generated YAML
  --click-verify {auto,off,yesno,diff,diff+yesno}
                        Override scenario's verify_click mode globally (A/B
                        testing)
  --canary {auto,off,on}
                        Override scenario's canary setting globally (A/B
                        testing)
  --no-record-session   Disable VNC frame recording (recording is ON by
                        default)
  --record-fps FPS      Recorder capture rate (default: 10.0)
  --record-diff-threshold VALUE
                        Mean grayscale diff threshold for frame retention
                        (default: 3.0)
  --record-gif          Also export recording/session.gif (MP4 is always
                        attempted)
  --record-keep-raw     Keep recording/raw frames after export
  --inject-config JSON  I1: JSON object (inline or @path.json) merged into
                        config.json before launch. Pins
                        __internal__.migrations.version and defaults
                        currentView from --inject-servers.
  --inject-servers LIST
                        I1: TITLE=URL,TITLE=URL (or @path.json with
                        {title:url} dict). Written as servers.json to both
                        userData dirs before launch.
  --inject-app-name NAME
                        I1: Electron productName for userData path detection
                        (default: auto-detect via strings(app.asar)).
  --inject-install-path PATH
                        I1: Override install path (binary or dir) for
                        asar/version detection. Default: derived from VM
                        package app_path.
  --inject-migrations-version VER
                        I1: pin __internal__.migrations.version explicitly
                        (default: auto-detect via <binary> --version).
  --no-cache            I6: disable VLM verify result cache for this run
  --config-snapshots    I14: capture config.json after each shell step
                        (overrides yaml; default off)
  --var KEY=VALUE       I8: override or set a scenario var (repeatable).
                        Format: KEY=VALUE. CLI overrides scenario yaml vars:
                        block. Error on duplicate keys within CLI --var flags.
  --from-phase ID       I9: start at the first step of named phase ID
                        (overrides --from-step; warns if both given)
  --until-phase ID      I9: stop after the last step of named phase ID
                        (overrides --until-step; warns if both given)
